## 🔹 Ansible — Install & Enable Nginx

---
- name : Install & Start Nginix on Bravo
  host : all
  become : yes

  tasls :
    - name : Install nginx
      yum :
        name : nginx
        state : present

    - name : Enable nginx
      service :
         name : nignx
         state : started
         enable : yes

## 🔹 Jenkins Pipeline — Maven Build

pipeline{
    agent any
    tools{
        mavern 'Maven-3.8.1'
        jdk 'JDK-17'
    }
    stages {
        stage('Checkout'){
            steps{
                git url : ''
            }
        }
        stage('Compile'){
            steps{
                sh 'mvn clean compile'
            }
        }
        stage('Test'){
            steps{
                sh 'mvn test'
            }
        }
        stage('Package'){
            steps{
                sh 'mvn package'
            }
        }    
    }
}


## 🔹 GitLab CI — Maven Build

stages :
 - build
 - test
 - package

maven-build :
  stage : build
  image : maven:3.8.1-jdk-17
  script:
    - git clone 
    - cd sample-java-app
    - mvn clean compile

maven-test :
   stage : test
   image : maven : 3.8.1-jdk-17
   script :
     - cd sample-java-app
     - mvn test

maven-package :
    stage : package
    image : maven:3.8.1-jdk-17
    script :
      - cd sample-java-app
      - mvn package
