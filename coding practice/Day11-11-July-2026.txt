# Day 11 — Database, Outputs, Sidecars, Functions

## 🔹 Ansible — Install MariaDB

---
- name : Install MariaDB for Pathnex
  hosts : all
  become : yes

  tasks:
   - name : Install MariaDB
     yum :
       name : mariadb-server
       state : present

    - name : Start MariaDB 
      service : 
        name : mariadb
        state : started
        enable : yes

## 🔹 Jenkins Pipeline — Email Notifications

pipeline{
    agent any
    environment{
        INSTITUTE_NAME = "Pathnex"
    }
    stages{
        stage('Checkout'){
            steps{
                git url : <git_repository>
            }
        }
        stage('Build'){
            steps{
                sh 'mvn clean package'
            }
        }
    }

    post{
        success{
            mail to : 'devTeam@pathnex.com',
                 subject : "Sucess : Build #${env.BUILD_NUMBER}",
                 body : "Build completed successfully for $INSTITUTE_NAME"
        }
        failure{
             mail to : 'devTeam@pathnex.com',
                 subject : "Sucess : Build #${env.BUILD_NUMBER}",
                 body : "Build completed successfully for $INSTITUTE_NAME"
        }
    }
}
