### Ansible:

---
- name: Create Pathnex Admin User
  hosts: all
  become: true

  vars:
    username: pathnex-admin

  tasks:

    - name: Create user
      ansible.builtin.user:
        name: "{{ username }}"
        shell: /bin/bash
        state: present
        create_home: true

    - name: Add SSH public key
      ansible.posix.authorized_key:
        user: "{{ username }}"
        key: "{{ lookup('file', '~/.ssh/id_rsa.pub') }}"
        state: present

## 5️⃣ Jenkins Pipeline

pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM           = "DevOps"
        ENV            = "prod"
        PROJECT        = "Pathnex-Training"
    }

    stages {

        stage('Infrastructure') {
            steps {
                echo "Creating ${PROJECT} infrastructure"

                dir('terraform') {
                    sh 'terraform init'
                    sh 'terraform validate'
                    sh 'terraform plan'
                    sh 'terraform apply -auto-approve'
                }
            }
        }

        stage('App Deployment') {
            steps {
                echo "Deploying application for ${ENV}"

                dir('ansible') {
                    sh 'ansible-playbook -i inventory.ini create-user.yml'
                }

                dir('kubernetes') {
                    sh 'kubectl apply -f app.yaml'
                }
            }
        }

        stage('Monitoring Checks') {
            steps {
                echo "Monitoring ${INSTITUTE_NAME} ${TEAM} environment"

                sh 'chmod +x monitoring.sh'
                sh './monitoring.sh'
            }
        }
    }

    post {
        success {
            echo "Pipeline completed successfully"
        }

        failure {
            echo "Pipeline failed"
        }
    }
}


## 6️⃣ GitLab CI

stages:
  - infrastructure
  - app_deployment
  - monitoring_checks

variables:
  INSTITUTE_NAME: "Pathnex"
  TEAM: "DevOps"
  ENV: "prod"
  PROJECT: "Pathnex-Training"

terraform:
  stage: infrastructure
  script:
    - echo "Creating $PROJECT infrastructure"
    - cd terraform
    - terraform init
    - terraform validate
    - terraform plan
    - terraform apply -auto-approve

ansible:
  stage: app_deployment
  script:
    - echo "Configuring server using Ansible"
    - cd ansible
    - ansible-playbook -i inventory.ini create-user.yml

kubernetes:
  stage: app_deployment
  script:
    - echo "Deploying application to Kubernetes"
    - cd kubernetes
    - kubectl apply -f app.yaml

monitoring:
  stage: monitoring_checks
  script:
    - echo "Monitoring $INSTITUTE_NAME $TEAM $ENV environment"
    - chmod +x monitoring.sh
    - ./monitoring.sh
