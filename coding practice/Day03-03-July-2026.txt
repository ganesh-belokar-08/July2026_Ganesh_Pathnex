## 🔹 Terraform — Security Group (r6i.4xlarge EC2)

provider "aws" {
    region = "us-eash-1"
}

resources "aws_security_group" "PathnexSG"{
    name : "Pathnex"
    description = "Pathnex SG for SSH"

    ingress {
        from_port = 22
        to_port = 22
        protocol = "tcp"
        cidr_blocks = ["0.0.0.0/0"]
    }
}

resources "aws_instance" "PathnexEC2"{
    ami = "ami-09934343sdresfds"
    instance_type = "r6i.4xlarge"
    vpc_security_group_ids = [aws_security_group.PathnexSG.id]

    tags= {
        Name = "Pathnex-EC2"
    }
}

# NodeJS Build
## 🔹 Jenkins Pipeline — NodeJS Build

pipeline {
    agent any
    tools{
        nodejs 'NodeJS-16'
    }
    stages {
        stage('Checkout'){
            steps{
                git url : ' '
            }
        }
    stage('Install Dependencies'){
        steps{
            sh 'npm install'
        }
    }
    stage('Build'){
        steps{
            sh 'npm run build'
        }
    }
    }
}
